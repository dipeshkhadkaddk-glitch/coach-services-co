async function loadDashboard() { // Bookings const bRes = await api.getBookings(); if (!bRes.ok) { toast.error('Failed to load bookings: ' + (bRes.data.message || 'Server error')); document.getElementById('bookings-table').innerHTML = '⚠️ Failed to load bookings'; } else { const bookings = bRes.data.data || []; document.getElementById('stat-total-bookings').textContent = bookings.length; document.getElementById('stat-confirmed').textContent = bookings.filter(b => b.status === 'confirmed').length; document.getElementById('stat-pending').textContent = bookings.filter(b => b.status === 'pending').length; const tbody = document.getElementById('bookings-table'); if (bookings.length === 0) { tbody.innerHTML = 'No bookings yet. Book now →'; } else { tbody.innerHTML = bookings.slice(0, 5).map(b => `
${b.pickup_location || 'N/A'}
→ ${b.dropoff_location || 'N/A'}
${statusBadge(b.booking_type)} ${statusBadge(b.status)} ${formatDate(b.created_at)} `).join(''); } } // Events const eRes = await api.getEvents(); const eventsEl = document.getElementById('events-list'); if (!eRes.ok) { document.getElementById('stat-events').textContent = '0'; eventsEl.innerHTML = '
⚠️ Failed to load events: ' + (eRes.data.message || 'Server error') + '
'; } else { const events = (eRes.data.data || []).slice(0, 3); document.getElementById('stat-events').textContent = eRes.data.data ? eRes.data.data.length : 0; if (events.length === 0) { eventsEl.innerHTML = '
🏅

No events scheduled yet

'; } else { eventsEl.innerHTML = events.map(ev => { const d = new Date(ev.event_date); return `
${d.getDate().toString().padStart(2,'0')}
${d.toLocaleDateString('en-AU',{month:'short'})}

${ev.name}

${ev.venue_name}
🕐${formatTime(ev.start_time)} – ${formatTime(ev.end_time)}
📍${ev.venue_address}
`; }).join(''); } } }
